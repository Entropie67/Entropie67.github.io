Le code complet du Puissance 4

Version finale, telle que programmée par la classe

Le cœur du jeu en Python — expliqué pas à pas, comme en classe.

puissance4.py Python · 4E5 
NB_LIGNES   = 6
NB_COLONNES = 7
VIDE    = 0
JOUEUR1 = 1
JOUEUR2 = 2

def creer_grille():                          # fabrique une grille vide de 6 lignes x 7 colonnes
    grille = []
    for ligne in range(NB_LIGNES):
        une_ligne = []
        for colonne in range(NB_COLONNES):
            une_ligne.append(VIDE)
        grille.append(une_ligne)
    return grille

def placer_pion(grille, colonne, joueur):    # fait "tomber" le pion tout en bas de la colonne
    for ligne in range(NB_LIGNES-1, -1, -1):
        if grille[ligne][colonne] == VIDE:
            grille[ligne][colonne] = joueur
            return

def grille_pleine(grille):                   # plus de place pour jouer ?
    for colonne in range(NB_COLONNES):
        if grille[0][colonne] == VIDE:
            return False
    return True

def verifier_victoire(grille, joueur):       # 4 pions alignés ? (horizontal, vertical, diagonales)
    for l in range(NB_LIGNES):
        for c in range(NB_COLONNES-3):
            if grille[l][c]==joueur and \
               grille[l][c+1]==joueur and \
               grille[l][c+2]==joueur and \
               grille[l][c+3]==joueur:
                return True
    # ... même logique pour : vertical,
    # diagonale montante, diagonale descendante
    return False

def jouer():                                 # boucle principale du jeu
    grille = creer_grille()
    joueur = JOUEUR1
    while True:
        colonne = int(input("Choisis une colonne : "))
        placer_pion(grille, colonne, joueur)
        if verifier_victoire(grille, joueur):
            # annonce du gagnant
            return
        if grille_pleine(grille):
            # match nul
            return
        joueur = JOUEUR2 if joueur==JOUEUR1 else JOUEUR1
